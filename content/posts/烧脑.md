## 位运算+dp+数学集合
### [n个数分最少的互质组](https://www.luogu.com.cn/problem/U207723)
#### 注意集合子集写法
##### 基本原理:
对每一种数字排列的组合进行一下判断：
* 列举此组合的所有子集，如果这个子集是互质的，那这个组合的次数就是这个组合删除这个子集的集合的次数加上1，然后对所有的子集重复操作取最小值即时最终答案
```c++
void solve()
{
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++)
    {
        cin >> a[i];
    }

    // 构造二维数组判断每两个数之间是否是互质的
    vector<vector<bool>> p(n, vector<bool>(n, false));
    for (int i = 0; i < n; i++)
    {
        for (int j = i + 1; j < n; j++)
        {
            if (gcd(a[i], a[j]) > 1)
            {
                p[i][j] = p[j][i] = true;
            }
        }
    }

    // 对于n个数中选取所有集合，判断这个集合是否是两两互质的，存储在q数组中
    int full = 1 << n;
    vector<char> q(full, 0);
    for (int mask = 0; mask < full; mask++)
    {
        bool ok = true;
        for (int i = 0; i < n; i++)
        {
            if (!((1 << i) & mask))
                continue;
            for (int j = i + 1; j < n; j++)
            {
                if (!((1 << j) & mask))
                    continue;
                if (p[i][j])
                {
                    ok = false;
                    break;
                }
            }
            q[mask] = ok ? 1 : 0;
        }
    }

    vector<int> dp(full, 1e9); // dp[i]数组表示覆盖i个数所需要的最少数量
    dp[0] = 0;
    for (int mask = 1; mask < full; mask++)
    {
        // 枚举每个mask的子集
        for (int sub = mask; sub; sub = (sub - 1) & mask)
        {
            if (q[sub])
            {
                dp[mask] = min(dp[mask], dp[mask ^ sub] + 1);
            }
        }
    }
    cout << dp[full - 1] << endl;
    return;
}
```
## 位运算+图基础
#### [求树中任意两点之间的路径长度（具有记忆性）](https://atcoder.jp/contests/abc376/tasks/abc376_d?lang=en)
```c++
const ll INF = (1ll << 60);

void solve()
{
    int n, Q;
    cin >> n >> Q;
    vector<vector<int>> g(n + 1); // 建立邻接矩阵表
    for (int i = 1; i < n; i++)
    {
        int a, b;
        cin >> a >> b;
        g[a].push_back(b);
        g[b].push_back(a);
    }

    int LOG = 1;
    while ((1 << LOG) <= n)
        ++LOG;

    vector<int> parent(n + 1, 0), depth(n + 1, 0);
    vector<vector<int>> up(n + 1, vector<int>(LOG, 0)); // up[v][k]表示节点v跳2的k次方层的祖先
    queue<int> q;
    q.push(1); // 1作为根节点，其他节点也可以
    parent[1] = 0;
    depth[1] = 0;
    up[1][0] = 0;
    while (!q.empty())
    {
        int x = q.front();
        q.pop(); // 取出队头元素
        for (int u : g[x])
        { // 遍历当前节点的所有邻接点
            if (u == parent[x])
            { // 如果是自己的父节点则跳过
                continue;
            }
            parent[u] = x;           // 将剩余节点的父节点设置为当前取出的节点
            depth[u] = depth[x] + 1; // 深度加一
            up[u][0] = x;            // 记录父节点
            q.push(u);               // 队列添加元素
        }
    }
    // 构造binary lifting表
    for (int i = 1; i < LOG; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            up[j][i] = up[up[j][i - 1]][i - 1];
        }
    }

    // lamdba函数求LCA
    auto lca = [&](int a, int b)
    {
        if (depth[a] < depth[b])
            swap(a, b);
        int diff = depth[a] - depth[b];
        for (int i = 0; i < LOG; i++)
        { // 将a和b拉到同一深度
            if (diff & (1 << LOG))
            {
                a = up[a][i];
            }
        }
        if (a == b)
        {
            return a;
        }
        for (int k = LOG - 1; k >= 0; k--)
        {
            if (up[a][k] != up[b][k])
            {
                a = up[a][k];
                b = up[b][k];
            }
        }
        return up[a][0];
    };
    while (Q--)
    {
        int c, d;
        cin >> c >> d;
        int L = lca(c, d);
        int dist = depth[c] + depth[d] - 2 * depth[L];
        cout << dist << endl;
    }
    return;
}
```
